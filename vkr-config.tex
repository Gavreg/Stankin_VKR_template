%должность, степ. , зван.
\def\insertPersonInfo#1{%
	\edef\tmpZ{\csname vkr#1Position\endcsname}%
	\edef\tmpA{\csname vkr#1DegreeShort\endcsname}%
	\edef\tmpB{\csname vkr#1TitleShort\endcsname}%
	\tmpZ%
	\ifx\tmpA\empty\else%
		,\space\tmpA%
	\fi%
	\ifx\tmpB\empty\else%
		,\space\tmpB%
	\fi%	
}
